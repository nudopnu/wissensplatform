program Stolperprotokoll

enum FootState
  NONE
  HEEL_STRIKE
  TOE_OFF
end

enum TrialState
  DEFAULT
  HABITUATION
  DETERMINE_STRIDE
  PROTOTOCOL
end

state
  left FootState.NONE
  right FootState.NONE
  trial TrialState.DEFAULT
  strides_tracker StrideTracker { capacity: 10 }
  avg_stride_left 0
  avg_stride_right 0
end

listener
    on left_foot_heel_strike
    when trial == TrialState.DETERMINE_STRIDE
    call strides_tracker.start_left
end

event left_foot_heel_strike
  when fz_left[-1] < -400 and left != FootState.HEEL_STRIKE
  set left HEEL_STRIKE
end

event stride_determination_completed
  when strides_tracker.has_completed
  set avg_stride_left  strides_tracker.avg_stride_left
  set avg_stride_right strides_tracker.avg_stride_right
  set trial TrialState.DEFAULT
  call strides_tracker.reset
  emit stride_determination_completed
end

button stride_determination_started "Determine Stride"
  set trial TrialState.DETERMINE_STRIDE
end

button habituation_started "Start Habituation"
  set trial TrialState.HABITUATION
end

program habituation
  wait 10000
  enqueue
end

sequence trip_left
  lerp 80 { lbs 1.0 }
  wait 120
  lerp 200 { lbs 0 }
end

sequence trip_crazy
  set { sway 0.05, lbs -3 }
  wait 60
  set { sway 0.05, lbs -3 }
  wait 60
end
